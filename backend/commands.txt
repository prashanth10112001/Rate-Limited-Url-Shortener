-------------------------- DATABASE CONNECTION SETUP ---------------------------------------------------

python -c '
import os
from dotenv import load_dotenv
from sqlalchemy import text
from sqlmodel import create_engine

load_dotenv()

db_url = os.getenv("DATABASE_URL") or (
    f"postgresql+psycopg2://{os.getenv('\''POSTGRES_USER'\'')}:{os.getenv('\''POSTGRES_PASSWORD'\'')}"
    f"@{os.getenv('\''POSTGRES_HOST'\'', '\''localhost'\'')}:{os.getenv('\''POSTGRES_PORT'\'', '\''5432'\'')}/{os.getenv('\''POSTGRES_DB'\'')}"
)

engine = create_engine(db_url, pool_pre_ping=True, pool_recycle=300)

with engine.connect() as conn:
    print("Connected:", conn.execute(
        text("SELECT current_database(), current_user")
    ).one())
'

from sqlalchemy import text

    print("Engine check:", conn.execute(text("SELECT 1")).scalar())

session = next(gen)
try:
    print("Session check:", session.exec(text("SELECT current_database()")).scalar())
finally:
    gen.close()
'
